-- 优惠券模板（由运营创建，代表一类优惠券）
CREATE TABLE t_coupons (
    id BIGSERIAL PRIMARY KEY,
    coupon_code VARCHAR(32) NOT NULL UNIQUE,
    name VARCHAR(64) NOT NULL,
    type SMALLINT NOT NULL DEFAULT 0,          -- 0=满减, 1=折扣
    amount NUMERIC(10,2) NOT NULL DEFAULT 0.00,-- 满减金额 或 折扣比例（如 0.9 = 9折）
    min_amount NUMERIC(10,2) NOT NULL DEFAULT 0.00, -- 使用门槛（最低消费）
    valid_start TIMESTAMP WITHOUT TIME ZONE NOT NULL,
    valid_end TIMESTAMP WITHOUT TIME ZONE NOT NULL,
    status SMALLINT NOT NULL DEFAULT 0,        -- 模板状态（通常不更新，保留字段）
    create_time TIMESTAMP WITHOUT TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    update_time TIMESTAMP WITHOUT TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    deleted BOOLEAN NOT NULL DEFAULT FALSE   -- 逻辑删除
);

COMMENT ON TABLE t_coupons IS '优惠券模板表';
COMMENT ON COLUMN t_coupons.coupon_code IS '优惠券码（唯一标识）';
COMMENT ON COLUMN t_coupons.type IS '类型：0=满减，1=折扣';
COMMENT ON COLUMN t_coupons.amount IS '减免金额（满减）或折扣比例（如0.9）';
COMMENT ON COLUMN t_coupons.min_amount IS '最低消费金额门槛';
CREATE INDEX idx_coupons_code ON t_coupons(coupon_code);
CREATE INDEX idx_coupons_valid_range ON t_coupons(valid_start, valid_end);
CREATE INDEX idx_coupons_deleted ON t_coupons(deleted);

-- 用户领取的优惠券实例（每个用户每张券一条记录）
CREATE TABLE t_user_coupons (
    id BIGSERIAL PRIMARY KEY,
    user_id BIGINT NOT NULL,
    coupon_id BIGINT NOT NULL REFERENCES t_coupons(id) ON DELETE CASCADE,
    code VARCHAR(32) NOT NULL,                 -- 冗余 coupon_code，避免连表
    status SMALLINT NOT NULL DEFAULT 0,        -- 0=未使用, 1=已使用, 2=已过期
    used_time TIMESTAMP WITHOUT TIME ZONE,
    create_time TIMESTAMP WITHOUT TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    update_time TIMESTAMP WITHOUT TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    deleted BOOLEAN NOT NULL DEFAULT FALSE
);

COMMENT ON TABLE t_user_coupons IS '用户优惠券领取记录表';
COMMENT ON COLUMN t_user_coupons.status IS '状态：0=未使用，1=已使用，2=已过期';
COMMENT ON COLUMN t_user_coupons.code IS '冗余存储 coupon_code，便于快速查询';

CREATE INDEX idx_user_coupons_user_id ON t_user_coupons(user_id);
CREATE INDEX idx_user_coupons_code ON t_user_coupons(code);
CREATE INDEX idx_user_coupons_status ON t_user_coupons(status);
CREATE INDEX idx_user_coupons_deleted ON t_user_coupons(deleted);
CREATE INDEX idx_user_coupons_user_status ON t_user_coupons(user_id, status) WHERE deleted = false;
